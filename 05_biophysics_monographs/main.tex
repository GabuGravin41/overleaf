\chapter*{How to Use This Platform}
\addcontentsline{toc}{chapter}{How to Use This Platform}

\noindent This platform is designed as a \textbf{Codeforces-equivalent for biology} —
a resource where students who have achieved gold medals at the IBO, and working researchers,
will still find the problems challenging. The material is not organised around rote memorisation.
Every concept is derived from physical first principles.

\medskip
\noindent\textbf{Each module contains:}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Deep lecture notes} --- full derivations, physical reasoning,
        historical context, and Python simulations.
  \item \textbf{Problem sets} --- 8--12 problems per module, each tagged
        with a difficulty rating and the disciplines involved.
  \item \textbf{Full worked solutions} --- every algebraic step shown.
\end{enumerate}

\subsection*{Difficulty rating}

\begin{tabular}{ll}
  \stars{1} & Accessible; requires careful reading of the notes. \\[3pt]
  \stars{2} & Requires synthesis of two or more concepts. \\[3pt]
  \stars{3} & Requires a derivation or quantitative argument from scratch. \\[3pt]
  \stars{4} & Challenging integration across topics; IBO gold level. \\[3pt]
  \stars{5} & Research-adjacent; open-ended or requires independent reasoning. \\
\end{tabular}
